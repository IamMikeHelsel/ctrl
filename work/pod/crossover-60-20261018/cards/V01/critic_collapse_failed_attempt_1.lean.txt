import Mathlib.Tactic
import Mathlib.LinearAlgebra.Matrix.Determinant.Basic
open Matrix
namespace V01Critic
/-- Concrete singular map, with both normalized ray pairs carrying positive depths. -/
theorem positive_depth_collapse :
    let H : Matrix (Fin 3) (Fin 3) ℝ := !![0,0,0;0,1,0;0,0,1]
    H.det = 0 ∧
    H *ᵥ ![1,0,1] = ![0,0,1] ∧
    H *ᵥ ![1/2,0,1] = ![0,0,1] ∧
    (![1,0,1] : Fin 3 → ℝ) ≠ ![1/2,0,1] ∧
    (0:ℝ)<1 ∧ (0:ℝ)<2 := by
  dsimp
  refine ⟨?_,?_,?_,?_,by norm_num,by norm_num⟩
  · norm_num [det_fin_three]
  · ext i; fin_cases i <;> norm_num [mulVec,dotProduct,Fin.sum_univ_three]
  · ext i; fin_cases i <;> norm_num [mulVec,dotProduct,Fin.sum_univ_three]
  · intro h
    have hi := congrFun h 0
    norm_num at hi

/-- The rays are generated by R=I,t=(-1,0,0) on the actual plane x=1. -/
theorem rigid_plane_binding :
    (1:ℝ) • (![1,0,1] : Fin 3 → ℝ) + (![-1,0,0] : Fin 3 → ℝ) = ![0,0,1] ∧
    (2:ℝ) • (![1/2,0,1] : Fin 3 → ℝ) + ![-1,0,0] = 2 • ![0,0,1] ∧
    (![1,0,0] : Fin 3 → ℝ) ⬝ᵥ ![1,0,1] = 1 ∧
    (![1,0,0] : Fin 3 → ℝ) ⬝ᵥ (2 • ![1/2,0,1]) = 1 := by
  constructor
  · ext i; fin_cases i <;> norm_num
  constructor
  · ext i; fin_cases i <;> norm_num
  constructor <;> norm_num [dotProduct,Fin.sum_univ_three]

/-- Invertibility is needed for reflection of determinant-defined collinearity. -/
theorem collinearity_reflection_fails :
    (Matrix.of ![![1,0,1],![0,1,1],![1,1,1]] : Matrix (Fin 3) (Fin 3) ℝ).det = -1 ∧
    (Matrix.of ![![1,0,0],![0,1,0],![1,1,0]] : Matrix (Fin 3) (Fin 3) ℝ).det = 0 := by
  norm_num [det_fin_three]
end V01Critic
